DELIMITER $$

CREATE TRIGGER add_theme
AFTER INSERT ON users
FOR EACH ROW
BEGIN
    DECLARE default_background VARCHAR(255);
    SET default_background = '_4e04ee6b-823f-4708-9517-5bbe1fecab8e.jpg';

    -- Verifica se o usuário inserido não tem um tema definido
    IF NEW.id IS NOT NULL AND NEW.name IS NOT NULL THEN
        INSERT INTO themes (id, user_id, name, primary_color_id, secondary_color_id, background, dark_theme)
        VALUES (DEFAULT, NEW.id, 'default', '#EAEAEA80', '#26260C', default_background, 0);
    END IF;
END$$

DELIMITER ;
